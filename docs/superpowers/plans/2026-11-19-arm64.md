# Arm64 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Sanduhr for Windows as native arm64 on both channels (a second Store package, a second GitHub installer with its own update feed), checked on GitHub's Arm runner, with a one-time notice for x64 GitHub installs running emulated on an Arm PC.

**Architecture:** The two build scripts take a runtime and stamp or name everything by architecture; arm64 packs into its own Velopack channel while x64 packs exactly as it does today. A new architecture gate runs after every publish. CI gains a `windows-11-arm` job; a separate Arm check workflow installs the real installers on Arm. The notice is a pure decision in Core, one native read and a small service in the app.

**Tech Stack:** .NET 10, C#, WPF, xunit, PowerShell (must run under Windows PowerShell 5.1 and pwsh 7), MSBuild, Velopack (`vpk` 1.2.0, library 0.0.1298), GitHub Actions.

**Spec:** `docs/superpowers/specs/2026-11-19-arm64-design.md`. Read it before any task.

```csharp
using System.Diagnostics;
using Xunit;

namespace Sanduhr.Tests;

/// <summary>
/// 5.0 spec 7: scripts/check-architecture.ps1 stops a build whose output holds a binary for the wrong processor.
/// The files here are synthetic: a DOS header, a PE signature and the CPU field, nothing else.
/// </summary>
public class ArchitectureGateTests : IDisposable
{
    private const ushort X64 = 0x8664;
    private const ushort Arm64 = 0xAA64;
    private const ushort Neutral = 0x014C;   // what managed, architecture-neutral files carry

    private readonly TempDir _root = new();

    public void Dispose() => _root.Dispose();

    private static string Script()
    {
        var dir = new DirectoryInfo(AppContext.BaseDirectory);
        while (dir is not null && !File.Exists(Path.Combine(dir.FullName, "src", "Sanduhr.App", "Package.appxmanifest")))
            dir = dir.Parent;
        Assert.NotNull(dir);
        return Path.Combine(dir!.FullName, "scripts", "check-architecture.ps1");
    }

    private void Pe(string relative, ushort machine)
    {
        var path = Path.Combine(_root.Path, relative);
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        var bytes = new byte[0x100];
        bytes[0] = (byte)'M';
        bytes[1] = (byte)'Z';
        BitConverter.GetBytes(0x80).CopyTo(bytes, 0x3C);
        bytes[0x80] = (byte)'P';
        bytes[0x81] = (byte)'E';
        BitConverter.GetBytes(machine).CopyTo(bytes, 0x84);
        File.WriteAllBytes(path, bytes);
    }

    private void GoodArm64()
    {
        Pe("Sanduhr.exe", Arm64);
        Pe(Path.Combine("mcp", "sanduhr-mcp.exe"), Arm64);
        Pe("coreclr.dll", Arm64);
        Pe("Sanduhr.Core.dll", Neutral);
    }

    private (int Exit, string Output) Run(string runtime)
    {
        var start = new ProcessStartInfo(StatuslineScriptRunTests.FindPowerShell()!)
        {
            RedirectStandardOutput = true,
            RedirectStandardError = true,
            UseShellExecute = false,
            CreateNoWindow = true,
        };
        foreach (var a in new[] { "-NoProfile", "-ExecutionPolicy", "Bypass", "-File", Script(), "-Dir", _root.Path, "-Runtime", runtime })
            start.ArgumentList.Add(a);
        using var p = Process.Start(start)!;
        var output = p.StandardOutput.ReadToEnd() + p.StandardError.ReadToEnd();
        p.WaitForExit();
        return (p.ExitCode, output);
    }

    [PowerShellFact]
    public void A_folder_of_the_right_architecture_passes()
    {
        GoodArm64();
        var (exit, output) = Run("win-arm64");
        Assert.True(exit == 0, output);
    }

    [PowerShellFact]
    public void A_file_for_the_other_64_bit_architecture_fails_and_is_named()
    {
        GoodArm64();
        Pe(Path.Combine("runtimes", "stray.dll"), X64);
        var (exit, output) = Run("win-arm64");
        Assert.Equal(1, exit);
        Assert.Contains("stray.dll", output);
    }

    [PowerShellFact]
    public void The_app_itself_must_be_the_target_architecture()
    {
        GoodArm64();
        Pe("Sanduhr.exe", Neutral);
        var (exit, output) = Run("win-arm64");
        Assert.Equal(1, exit);
        Assert.Contains("Sanduhr.exe", output);
    }

    [PowerShellFact]
    public void A_missing_server_fails()
    {
        GoodArm64();
        File.Delete(Path.Combine(_root.Path, "mcp", "sanduhr-mcp.exe"));
        var (exit, output) = Run("win-arm64");
        Assert.Equal(1, exit);
        Assert.Contains("sanduhr-mcp.exe", output);
    }

    [PowerShellFact]
    public void A_dll_that_is_not_a_program_file_is_ignored()
    {
        GoodArm64();
        File.WriteAllText(Path.Combine(_root.Path, "notes.dll"), "not a binary");
        var (exit, output) = Run("win-arm64");
        Assert.True(exit == 0, output);
    }

    [PowerShellFact]
    public void An_x64_folder_with_an_arm64_stray_fails()
    {
        Pe("Sanduhr.exe", X64);
        Pe(Path.Combine("mcp", "sanduhr-mcp.exe"), X64);
        Pe("stray.dll", Arm64);
        var (exit, output) = Run("win-x64");
        Assert.Equal(1, exit);
        Assert.Contains("stray.dll", output);
    }

    [PowerShellFact]
    public void An_unknown_runtime_is_refused()
    {
        GoodArm64();
        var (exit, _) = Run("win-x86");
        Assert.NotEqual(0, exit);
    }
}
```

```powershell
<#
.SYNOPSIS
  Fail when a publish folder holds a program file built for the wrong processor.

.DESCRIPTION
  Sanduhr ships for win-x64 and win-arm64. A build machine can run only one of them, so a
  wrong binary would otherwise be found by a user. This reads the CPU field in the header of
  every .dll and .exe under -Dir and fails when:
    - Sanduhr.exe or mcp\sanduhr-mcp.exe is missing or is not built for -Runtime, or
    - any file is built for the other 64-bit architecture.
  Managed, architecture-neutral files (the field reads i386) pass, and so does a .dll that is
  not a program file at all.

.EXAMPLE
  pwsh windows-dotnet/scripts/check-architecture.ps1 -Dir windows-dotnet/dist/publish-arm64 -Runtime win-arm64
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory)][string]$Dir,
    [Parameter(Mandatory)][ValidateSet('win-x64', 'win-arm64')][string]$Runtime
)

$ErrorActionPreference = 'Stop'
$Dir = [System.IO.Path]::GetFullPath($Dir).TrimEnd('\')
if (-not (Test-Path -LiteralPath $Dir -PathType Container)) { throw "No folder at $Dir." }

if ($Runtime -eq 'win-arm64') { $want = 'arm64'; $other = 'x64' } else { $want = 'x64'; $other = 'arm64' }

function Get-Machine {
    param([string]$Path)
    $fs = [System.IO.File]::OpenRead($Path)
    try {
        if ($fs.Length -lt 0x40) { return 'none' }
        $br = New-Object System.IO.BinaryReader($fs)
        if ($br.ReadUInt16() -ne 0x5A4D) { return 'none' }          # "MZ"
        $fs.Position = 0x3C
        $pe = $br.ReadInt32()
        if ($pe -le 0 -or ($pe + 6) -gt $fs.Length) { return 'none' }
        $fs.Position = $pe
        if ($br.ReadUInt32() -ne 0x00004550) { return 'none' }      # "PE\0\0"
        $machine = [int]$br.ReadUInt16()
        if ($machine -eq 0x8664) { return 'x64' }
        if ($machine -eq 0xAA64) { return 'arm64' }
        if ($machine -eq 0x014C) { return 'neutral' }
        return 'other'
    } finally { $fs.Dispose() }
}

$problems = @()
$counts = @{}
$files = @(Get-ChildItem -LiteralPath $Dir -Recurse -File | Where-Object { $_.Extension -eq '.dll' -or $_.Extension -eq '.exe' })
foreach ($f in $files) {
    $m = Get-Machine $f.FullName
    $counts[$m] = 1 + [int]$counts[$m]
    if ($m -eq $other) { $problems += ('{0} is built for {1}' -f $f.FullName.Substring($Dir.Length + 1), $other) }
}
foreach ($must in @('Sanduhr.exe', 'mcp\sanduhr-mcp.exe')) {
    $p = Join-Path $Dir $must
    if (-not (Test-Path -LiteralPath $p)) { $problems += "$must is missing"; continue }
    $m = Get-Machine $p
    if ($m -ne $want) { $problems += "$must is $m, not $want" }
}

$summary = ($counts.GetEnumerator() | Sort-Object Name | ForEach-Object { '{0} {1}' -f $_.Value, $_.Name }) -join ', '
if ($problems.Count -gt 0) {
    Write-Output "[architecture] FAIL for $Runtime in $Dir ($summary):"
    $problems | ForEach-Object { Write-Output "  - $_" }
    exit 1
}
Write-Output "[architecture] OK for $Runtime ($summary)."
exit 0
```

```csharp
using System.Runtime.InteropServices;
using System.Runtime.Versioning;

namespace Sanduhr.App.Interop;

/// <summary>
/// The machine's real processor architecture (5.0 spec 7). An x64 Sanduhr on an Arm PC runs through Windows'
/// emulation, and IsWow64Process2 is the call that tells an emulated process what the machine really is.
/// Hand-written <c>[DllImport]</c>, same pattern as <see cref="RestartNative"/>. Windows 10 1709 and later;
/// Sanduhr's floor is 1809.
/// </summary>
[SupportedOSPlatform("windows")]
internal static class MachineNative
{
    private const ushort IMAGE_FILE_MACHINE_I386 = 0x014C;
    private const ushort IMAGE_FILE_MACHINE_AMD64 = 0x8664;
    private const ushort IMAGE_FILE_MACHINE_ARM64 = 0xAA64;

    [DllImport("kernel32.dll")]
    private static extern IntPtr GetCurrentProcess();

    [DllImport("kernel32.dll", SetLastError = true)]
    [return: MarshalAs(UnmanagedType.Bool)]
    private static extern bool IsWow64Process2(IntPtr process, out ushort processMachine, out ushort nativeMachine);

    /// <summary>The machine's architecture, or null when Windows would not say.</summary>
    public static Architecture? Native()
    {
        try
        {
            if (!IsWow64Process2(GetCurrentProcess(), out _, out var native))
                return null;
            return native switch
            {
                IMAGE_FILE_MACHINE_ARM64 => Architecture.Arm64,
                IMAGE_FILE_MACHINE_AMD64 => Architecture.X64,
                IMAGE_FILE_MACHINE_I386 => Architecture.X86,
                _ => null,
            };
        }
        catch (Exception e) when (e is DllNotFoundException or EntryPointNotFoundException)
        {
            return null;
        }
    }
}
```

```csharp
using System.Diagnostics;
using System.Runtime.InteropServices;
using Microsoft.Toolkit.Uwp.Notifications;
using Sanduhr.App.Interop;
using Sanduhr.Core;

namespace Sanduhr.App.Services;

/// <summary>
/// The Arm build notice, on screen (5.0 spec 7). <see cref="ArmBuildNotice"/> decides; this reads the machine,
/// shows the one notification, remembers that it did, and opens the release list when its button is clicked.
/// The log line is written when the notification is due, before it is shown, so the Arm check workflow can see
/// the decision on a runner that cannot show a notification. Nothing here throws: a notice is never worth a
/// failed launch.
/// </summary>
internal sealed class ArmBuildNoticeService
{
    private readonly bool _applies;
    private readonly string _url;
    private readonly Func<bool> _alreadyShown;
    private readonly Action _markShown;
    private readonly Action _showToast;
    private readonly Action<string> _openLink;
    private readonly Action<string>? _log;

    public ArmBuildNoticeService(bool applies, string url, Func<bool> alreadyShown, Action markShown,
        Action? showToast = null, Action<string>? openLink = null, Action<string>? log = null)
    {
        _applies = applies;
        _url = url;
        _alreadyShown = alreadyShown;
        _markShown = markShown;
        _showToast = showToast ?? ShowToast;
        _openLink = openLink ?? OpenInBrowser;
        _log = log;
    }

    /// <summary>True when this copy is an x64 build on an Arm PC and is not the Store package.</summary>
    public static bool AppliesHere()
        => ArmBuildNotice.Applies(RuntimeInformation.ProcessArchitecture, MachineNative.Native(), Startup.IsPackaged());

    private static int Major() => System.Reflection.Assembly.GetEntryAssembly()?.GetName().Version?.Major ?? 0;

    /// <summary>The release list to get the Arm build from, or null when the notice does not apply here.</summary>
    public static string? UrlIfApplies() => AppliesHere() ? ArmBuildNotice.ReleasesUrl(Major()) : null;

    public static ArmBuildNoticeService ForThisApp(SettingsStore store, Action<string>? log)
        => new(AppliesHere(), ArmBuildNotice.ReleasesUrl(Major()), store.LoadArmBuildNoticeShown, store.SaveArmBuildNoticeShown, log: log);

    /// <summary>Hooks the notification's button. Does nothing where the notice does not apply.</summary>
    public void Start()
    {
        if (!_applies)
            return;
        try { Toasts.OnActivated(OnToast); }
        catch (Exception e) { Log($"activation-hook ({e.GetType().Name})"); }
    }

    /// <summary>A clicked notification's argument: only this notice's button opens the link.</summary>
    internal void OnToast(string argument)
    {
        if (argument.Contains(ArmBuildNotice.GetAction, StringComparison.Ordinal))
            _openLink(_url);
    }

    /// <summary>Shows the notification when it is due. True when it was handed to Windows.</summary>
    public bool ShowIfDue()
    {
        bool shown;
        try { shown = _alreadyShown(); }
        catch (Exception e)
        {
            Log($"read ({e.GetType().Name})");
            return false;
        }
        if (!ArmBuildNotice.ToastDue(_applies, shown))
            return false;
        _log?.Invoke(ArmBuildNotice.LogDue);
        try { _showToast(); }
        catch (Exception e)
        {
            // Not marked: the next launch tries again.
            Log($"toast ({e.GetType().Name})");
            return false;
        }
        try { _markShown(); }
        catch (Exception e) { Log($"flag ({e.GetType().Name})"); }
        return true;
    }

    private static void ShowToast() => Toasts.Show(ArmBuildNotice.Title, ArmBuildNotice.Body, more: b => b
        .AddButton(new ToastButton().SetContent(ArmBuildNotice.GetText).AddArgument("action", ArmBuildNotice.GetAction))
        .AddButton(new ToastButtonDismiss(ArmBuildNotice.LaterText)));

    private static void OpenInBrowser(string url)
    {
        if (!url.StartsWith("https://", StringComparison.Ordinal))
            return;
        try
        {
            // network: your-browser
            Process.Start(new ProcessStartInfo(url) { UseShellExecute = true })?.Dispose();
        }
        catch (Exception)
        {
            // The link is also in Settings, About, Updates.
        }
    }

    private void Log(string message) => _log?.Invoke("arm build notice: " + message);
}
```

```powershell
<#
.SYNOPSIS
  Start a published Sanduhr.exe, confirm it shows a window and stays up, then close it.

.DESCRIPTION
  The closest thing to "someone opened it" a runner can do. It starts the app in demo mode
  (sample numbers, nothing fetched), waits for a visible window, holds, and fails when the
  process exits early, never shows a window, or leaves a crash in the Application event log.

  CI RUNNERS ONLY. A Sanduhr started from a publish folder uses the same data folders as an
  installed Sanduhr. Never run this on a PC that has Sanduhr installed.

.PARAMETER NoWindowCheck
  For a runner that cannot show a window: pass on the process staying up alone.

.EXAMPLE
  pwsh windows-dotnet/scripts/smoke-launch.ps1 -Exe windows-dotnet/dist/publish-arm64/Sanduhr.exe
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory)][string]$Exe,
    [int]$HoldSeconds = 15,
    [int]$WindowTimeoutSeconds = 60,
    [switch]$NoWindowCheck
)

$ErrorActionPreference = 'Stop'
$Exe = [System.IO.Path]::GetFullPath($Exe)
if (-not (Test-Path -LiteralPath $Exe)) { throw "No app at $Exe." }

# Sanduhr is single-instance: a second copy hands off to the first and exits, which would read as a crash here.
if (Get-Process -Name Sanduhr -ErrorAction SilentlyContinue) { throw 'A Sanduhr is already running. Close it first.' }

Add-Type -Namespace SmokeLaunch -Name Win -MemberDefinition @'
public delegate bool EnumProc(System.IntPtr h, System.IntPtr l);
[DllImport("user32.dll")] public static extern bool EnumWindows(EnumProc cb, System.IntPtr l);
[DllImport("user32.dll")] public static extern uint GetWindowThreadProcessId(System.IntPtr h, out uint pid);
[DllImport("user32.dll")] public static extern bool IsWindowVisible(System.IntPtr h);
public static int VisibleWindows(uint pid) {
    int n = 0;
    EnumWindows(delegate(System.IntPtr h, System.IntPtr l) {
        uint p;
        GetWindowThreadProcessId(h, out p);
        if (p == pid && IsWindowVisible(h)) n++;
        return true;
    }, System.IntPtr.Zero);
    return n;
}
'@

$started = Get-Date
$p = Start-Process -FilePath $Exe -ArgumentList '--demo' -PassThru
try {
    $deadline = (Get-Date).AddSeconds($WindowTimeoutSeconds)
    $seen = $false
    while ((Get-Date) -lt $deadline) {
        $p.Refresh()
        if ($p.HasExited) { throw "Sanduhr exited during startup (code $($p.ExitCode))." }
        if ($NoWindowCheck -or ([SmokeLaunch.Win]::VisibleWindows([uint32]$p.Id) -gt 0)) { $seen = $true; break }
        [System.Threading.Thread]::Sleep(500)
    }
    if (-not $seen) { throw "Sanduhr showed no window within $WindowTimeoutSeconds seconds." }

    [System.Threading.Thread]::Sleep($HoldSeconds * 1000)
    $p.Refresh()
    if ($p.HasExited) { throw "Sanduhr exited within $HoldSeconds seconds of starting (code $($p.ExitCode))." }
    if (-not $NoWindowCheck -and ([SmokeLaunch.Win]::VisibleWindows([uint32]$p.Id) -eq 0)) { throw "Sanduhr's window went away." }
}
finally {
    $p.Refresh()
    if (-not $p.HasExited) { & taskkill.exe /PID $p.Id /T /F | Out-Null }
}

# A hard crash leaves an event even when a dialog kept the process alive.
$crashes = @(Get-WinEvent -FilterHashtable @{ LogName = 'Application'; Level = 2; StartTime = $started } -ErrorAction SilentlyContinue |
    Where-Object { ($_.ProviderName -eq '.NET Runtime' -or $_.ProviderName -eq 'Application Error') -and $_.Message -like '*Sanduhr.exe*' })
if ($crashes.Count -gt 0) {
    $first = ($crashes[0].Message -split "`n" | Select-Object -First 6) -join "`n"
    throw "Sanduhr left $($crashes.Count) crash event(s). First:`n$first"
}

Write-Output "[launch] OK: $Exe started, showed a window and stayed up $HoldSeconds seconds."
```

Create `windows-dotnet/scripts/check-release-assets.ps1`:

```powershell
<#
.SYNOPSIS
  Fail when a folder of release files is missing something a Windows release must carry.

.DESCRIPTION
  A Windows release ships two Store packages, two installers, two full update packages, two
  portable zips and two update feeds. An architecture whose feed is missing never updates, and
  nothing else notices. This checks that every file is there and that each feed lists its own
  architecture's full package at this version. Delta packages are optional and not checked.

  The tag may be three-part (v5.0.0) or four-part (v5.0.0.0). Velopack's file names carry the
  three-part version; the Store packages carry the four-part one.

.EXAMPLE
  pwsh windows-dotnet/scripts/check-release-assets.ps1 -Dir draft-files -Tag v5.0.0.0
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory)][string]$Dir,
    [Parameter(Mandatory)][ValidatePattern('^v\d+\.\d+\.\d+(\.\d+)?$')][string]$Tag
)

$ErrorActionPreference = 'Stop'
$Dir = [System.IO.Path]::GetFullPath($Dir)
if (-not (Test-Path -LiteralPath $Dir -PathType Container)) { throw "No folder at $Dir." }

$parts = ($Tag -replace '^v', '').Split('.')
$pack = $parts[0..2] -join '.'
if ($parts.Count -eq 4) { $store = $parts -join '.' } else { $store = "$pack.0" }

$x64Full = "626Labs.Sanduhr-${pack}-full.nupkg"
$armFull = "626Labs.Sanduhr-${pack}-win-arm64-full.nupkg"
$expected = @(
    "Sanduhr-Store-x64-v${store}.msix"
    "Sanduhr-Store-arm64-v${store}.msix"
    '626Labs.Sanduhr-win-Setup.exe'
    '626Labs.Sanduhr-win-arm64-Setup.exe'
    $x64Full
    $armFull
    '626Labs.Sanduhr-win-Portable.zip'
    '626Labs.Sanduhr-win-arm64-Portable.zip'
    'releases.win.json'
    'releases.win-arm64.json'
)

$problems = @()
foreach ($name in $expected) {
    if (-not (Test-Path -LiteralPath (Join-Path $Dir $name))) { $problems += "$name is missing" }
}

# Each feed must list its own architecture's full package at this version.
foreach ($feed in @(@('releases.win.json', $x64Full), @('releases.win-arm64.json', $armFull))) {
    $path = Join-Path $Dir $feed[0]
    if (-not (Test-Path -LiteralPath $path)) { continue }
    $names = @()
    try { $names = @((Get-Content -LiteralPath $path -Raw | ConvertFrom-Json).Assets | ForEach-Object { $_.FileName }) }
    catch { $problems += "$($feed[0]) is not readable JSON"; continue }
    if ($names -notcontains $feed[1]) { $problems += "$($feed[0]) does not list $($feed[1])" }
}

if ($problems.Count -gt 0) {
    Write-Output "[assets] FAIL for $Tag in ${Dir}:"
    $problems | ForEach-Object { Write-Output "  - $_" }
    exit 1
}
Write-Output "[assets] OK for ${Tag}: $($expected.Count) files, both feeds list their packages."
exit 0
```

```powershell
<#
.SYNOPSIS
  On an Arm runner: install the x64 Sanduhr, then the arm64 one over it, and check the switch.

.DESCRIPTION
  An Arm PC that installed Sanduhr from GitHub before 5.0 has the x64 build. This is what that
  person does to get the native one, run for real:
    1. Install the x64 installer. On an Arm PC that Sanduhr runs through emulation, so it must
       record that the Arm build notice was due.
    2. Put a marker file in each of Sanduhr's data folders.
    3. Install the arm64 installer (in place, or after an uninstall with -UninstallFirst).
       The installed app must be arm64, its package's channel must be win-arm64 (the feed it
       will update from), the markers must still be there, the app must start, and it must not
       record the notice as due.

  CI RUNNERS ONLY. It installs and replaces Sanduhr, kills running copies and writes into the
  real data folders. Never run it on a PC that has Sanduhr installed; it refuses to start there.

.EXAMPLE
  pwsh windows-dotnet/scripts/check-install-switch.ps1 -X64Setup dist/release/626Labs.Sanduhr-win-Setup.exe -Arm64Setup dist/release/626Labs.Sanduhr-win-arm64-Setup.exe
#>
[CmdletBinding()]
param(
    [Parameter(Mandatory)][string]$X64Setup,
    [Parameter(Mandatory)][string]$Arm64Setup,
    [switch]$UninstallFirst
)

$ErrorActionPreference = 'Stop'
$X64Setup = (Resolve-Path -LiteralPath $X64Setup).Path
$Arm64Setup = (Resolve-Path -LiteralPath $Arm64Setup).Path

$installRoot = Join-Path $env:LOCALAPPDATA '626Labs.Sanduhr'
$current = Join-Path $installRoot 'current'
$exe = Join-Path $current 'Sanduhr.exe'
$dataDirs = @((Join-Path $env:APPDATA 'Sanduhr'), (Join-Path $env:LOCALAPPDATA 'Sanduhr'), (Join-Path $env:USERPROFILE '.sanduhr'))
$logFile = Join-Path $env:APPDATA 'Sanduhr\sanduhr.log'
$dueLine = 'arm build notice: due'
$onArm = ($env:PROCESSOR_ARCHITECTURE -eq 'ARM64') -or ($env:PROCESSOR_ARCHITEW6432 -eq 'ARM64')

function Stop-Sanduhr {
    foreach ($proc in @(Get-Process -Name Sanduhr -ErrorAction SilentlyContinue)) { & taskkill.exe /PID $proc.Id /T /F | Out-Null }
    [System.Threading.Thread]::Sleep(2000)
}

function Wait-ForExit {
    param($Process, [int]$Seconds, [string]$What)
    $null = $Process.Handle   # hold the handle, or ExitCode is empty once the process has gone
    if (-not $Process.WaitForExit($Seconds * 1000)) {
        & taskkill.exe /PID $Process.Id /T /F | Out-Null
        throw "$What did not finish in $Seconds seconds."
    }
    if ($Process.ExitCode -ne 0) { throw "$What exited with code $($Process.ExitCode)." }
}

function Install-Setup {
    param([string]$Setup)
    $name = [System.IO.Path]::GetFileName($Setup)
    Write-Output "[switch] installing $name"
    Wait-ForExit (Start-Process -FilePath $Setup -ArgumentList '--silent' -PassThru) 300 $name
    $deadline = (Get-Date).AddSeconds(60)
    while (-not (Test-Path -LiteralPath $exe) -and ((Get-Date) -lt $deadline)) { [System.Threading.Thread]::Sleep(1000) }
    if (-not (Test-Path -LiteralPath $exe)) { throw "No $exe after $name." }
    # Setup may start the app when it finishes; the documentation does not say. Give a started app time to
    # settle and say so, then stop it, so every launch below is this script's own.
    [System.Threading.Thread]::Sleep(10000)
    if (Get-Process -Name Sanduhr -ErrorAction SilentlyContinue) { Write-Output "[switch] note: $name started the app after a silent install" }
    else { Write-Output "[switch] note: $name did not start the app after a silent install" }
    Stop-Sanduhr
}

function Assert-Installed {
    param([string]$Runtime, [string]$Channel)
    & (Join-Path $PSScriptRoot 'check-architecture.ps1') -Dir $current -Runtime $Runtime
    if ($LASTEXITCODE -ne 0) { throw "The installed copy is not $Runtime." }
    [xml]$record = Get-Content -LiteralPath (Join-Path $current 'sq.version') -Raw
    $meta = $record.package.metadata
    if ($meta.channel -ne $Channel) { throw "The installed package's channel is '$($meta.channel)', not '$Channel'." }
    Write-Output "[switch] installed: version $($meta.version), channel $($meta.channel)"
}

function Test-NoticeDue {
    return (Test-Path -LiteralPath $logFile) -and [bool](Select-String -LiteralPath $logFile -SimpleMatch $dueLine -Quiet)
}

Stop-Sanduhr
if (Test-Path -LiteralPath $installRoot) { throw "Sanduhr is already installed at $installRoot. This check is for a clean runner." }

# 1. The x64 build. On an Arm PC it runs through emulation and must say the notice was due.
Install-Setup $X64Setup
Assert-Installed -Runtime 'win-x64' -Channel 'win'
& (Join-Path $PSScriptRoot 'smoke-launch.ps1') -Exe $exe
if ($onArm) {
    if (-not (Test-NoticeDue)) { throw 'The x64 Sanduhr on an Arm PC did not record that the Arm build notice was due.' }
    Write-Output '[switch] the x64 build on Arm recorded the notice as due'
} else {
    Write-Output '[switch] not an Arm machine: the notice is not checked'
}

# 2. Markers in every data folder, and a fresh log for the arm64 build.
foreach ($d in $dataDirs) {
    New-Item -ItemType Directory -Path $d -Force | Out-Null
    Set-Content -LiteralPath (Join-Path $d 'arm-check-marker.txt') -Value 'marker'
}
if (Test-Path -LiteralPath $logFile) { Remove-Item -LiteralPath $logFile -Force }

# 3. The arm64 build over it.
if ($UninstallFirst) {
    Write-Output '[switch] uninstalling the x64 build first'
    Wait-ForExit (Start-Process -FilePath (Join-Path $installRoot 'Update.exe') -ArgumentList 'uninstall', '--silent' -PassThru) 120 'Update.exe uninstall'
    $deadline = (Get-Date).AddSeconds(60)
    while ((Test-Path -LiteralPath $current) -and ((Get-Date) -lt $deadline)) { [System.Threading.Thread]::Sleep(1000) }
    if (Test-Path -LiteralPath $current) { throw 'The x64 build is still installed after the uninstall.' }
}
Install-Setup $Arm64Setup
Assert-Installed -Runtime 'win-arm64' -Channel 'win-arm64'
foreach ($d in $dataDirs) {
    if (-not (Test-Path -LiteralPath (Join-Path $d 'arm-check-marker.txt'))) { throw "The marker in $d did not survive the switch." }
}
Write-Output '[switch] every data folder kept its marker'
& (Join-Path $PSScriptRoot 'smoke-launch.ps1') -Exe $exe
if (Test-NoticeDue) { throw 'The arm64 build recorded the Arm build notice as due. It must stay silent on a native build.' }

if ($UninstallFirst) { $how = 'after an uninstall' } else { $how = 'in place' }
Write-Output "[switch] OK: x64 to arm64 $how."
```
